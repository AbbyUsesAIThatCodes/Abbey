cmake_minimum_required(VERSION 3.22)
project(Abbey VERSION 0.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(ABBEY_BUILD_GODOT "Build the Godot GDExtension" OFF)
option(ABBEY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(abbey_core core/src/simulation.cpp)
target_include_directories(abbey_core PUBLIC core/include)
set_target_properties(abbey_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(abbey_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(abbey_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(ABBEY_SANITIZE AND NOT MSVC)
  target_compile_options(abbey_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(abbey_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(abbey_benchmark core/src/benchmark.cpp)
target_link_libraries(abbey_benchmark PRIVATE abbey_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(abbey_tests core/tests/simulation_tests.cpp)
  target_link_libraries(abbey_tests PRIVATE abbey_core)
  add_test(NAME simulation COMMAND abbey_tests)
endif()
if(ABBEY_BUILD_GODOT)
  include(FetchContent)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/native/build_profile.json" CACHE PATH "" FORCE)
  set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
  FetchContent_Declare(godot-cpp
    GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git
    GIT_TAG e4b7c25e721ce3435a029087e3917a30aa73f06b)
  FetchContent_MakeAvailable(godot-cpp)
  add_library(abbey_native SHARED native/abbey_bridge.cpp)
  target_link_libraries(abbey_native PRIVATE abbey_core godot-cpp)
  set_target_properties(abbey_native PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/godot/bin/$<0:>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/godot/bin/$<0:>")
endif()
